Metamodel Analysis and Humanoid Robot Application

Discussion Topic: OO Design for IoT

Referring to the article by Baskara et al (2024), consider the strengths and weaknesses of designing a metamodel to support object-oriented design of the IoT.

Analysis of Baskara et al (2024) Metamodel Approach

Strengths of IoT Metamodel Design

  • Abstraction and Standardization: Provides a unified framework for diverse IoT devices and protocols
  • Interoperability: Enables communication between heterogeneous IoT systems
  • Scalability: Supports addition of new device types and protocols
  • Maintainability: Centralized model management reduces code duplication
  • Reusability: Common patterns can be applied across different IoT domains

Weaknesses and Challenges

  • Complexity Overhead: Additional abstraction layers can increase system complexity
  • Performance Impact: Meta-level operations may introduce latency in real-time systems
  • Learning Curve: Developers need to understand both domain and metamodel concepts
  • Over-engineering Risk: May create unnecessary abstraction for simple IoT applications
  • Tooling Dependency: Often requires specialized modeling tools and expertise

class HumanoidRobotMetaModel:
    """Metamodel for humanoid robot systems based on Baskara et al approach."""
    
    def __init__(self):
        self.robot_types = {}
        self.capability_models = {}
        self.behavior_patterns = {}
        self.sensor_configs = {}
    
    def register_robot_type(self, robot_type):
        """Register a new robot type in the metamodel."""
        self.robot_types[robot_type.name] = robot_type
    
    def define_capability(self, capability):
        """Define a capability that robots can implement."""
        self.capability_models[capability.type] = capability
    
    def add_behavior_pattern(self, pattern):
        """Add a reusable behavior pattern."""
        self.behavior_patterns[pattern.name] = pattern
    
    def configure_sensor(self, sensor_config):
        """Configure sensor specifications."""
        self.sensor_configs[sensor_config.sensor_type] = sensor_config
    
    def generate_robot_instance(self, robot_spec):
        """Generate a specific robot instance from specifications."""
        robot_type = self.robot_types.get(robot_spec.robot_type)
        if not robot_type:
            raise ValueError(f"Unknown robot type: {robot_spec.robot_type}")
        
        return RobotInstance(robot_type, robot_spec.capabilities)

class RobotType:
    def __init__(self, name, capabilities, sensors, actuators):
        self.name = name
        self.capabilities = capabilities
        self.sensors = sensors
        self.actuators = actuators

class CapabilityModel:
    def __init__(self, capability_type, parameters, constraints, outcomes):
        self.type = capability_type
        self.parameters = parameters
        self.constraints = constraints
        self.outcomes = outcomes

class Card:

        """Convert rank to numeric value for comparison."""
        return Card.RANKS.index(self.rank) + 2

Exercise 1: Trick Winner Finder

class Trick(Deck):
    """Represents a trick in contract bridge."""
    
    def find_winner(self):
        """Find the index of the winning card in the trick."""
        if not self.cards:
            return -1
        
        # First card determines the led suit
        led_suit = self.cards[0].suit
        winning_index = 0
        highest_rank = self.cards[0].get_numeric_rank()
        
        # Check only cards of the led suit
        for i in range(1, len(self.cards)):
            card = self.cards[i]
            if card.suit == led_suit:
                card_rank = card.get_numeric_rank()
                if card_rank > highest_rank:
                    highest_rank = card_rank
                    winning_index = i
        
        return winning_index

Exercise 2-6: Poker Hand Classification

class PokerHand(Hand):
    """Represents a poker hand with classification methods."""
    
    def get_suit_counts(self):
        counter = {}
        for card in self.cards:
            key = card.suit
            counter[key] = counter.get(key, 0) + 1
        return counter
            
    def get_rank_counts(self):
        counter = {}
        for card in self.cards:
            key = card.rank
            counter[key] = counter.get(key, 0) + 1
        return counter
    
    def has_flush(self):
        """Check if hand has at least five cards of the same suit."""
        suit_counts = self.get_suit_counts()
        return any(count >= 5 for count in suit_counts.values())
    
    def has_straight(self):
        """Check if hand contains five cards with consecutive ranks."""
        numeric_ranks = sorted([card.get_numeric_rank() for card in self.cards])
        
        # Handle Ace as both high and low
        if 14 in numeric_ranks:  # Ace
            numeric_ranks.append(1)  # Add Ace as low
        
        # Remove duplicates and sort
        unique_ranks = sorted(set(numeric_ranks))
        
        # Check for consecutive sequence of 5
        for i in range(len(unique_ranks) - 4):
            if unique_ranks[i+4] - unique_ranks[i] == 4:
                return True
        
        return False
    
    def has_straight_flush(self):
        """Check if hand has both straight and flush."""
        if not self.has_flush():
            return False
        
        # Group cards by suit
        suit_groups = {}
        for card in self.cards:
            if card.suit not in suit_groups:
                suit_groups[card.suit] = []
            suit_groups[card.suit].append(card)
        
        # Check each suit group for straight
        for suit, cards in suit_groups.items():
            if len(cards) >= 5:
                temp_hand = PokerHand(cards)
                if temp_hand.has_straight():
                    return True
        
        return False
    
    def has_pair(self):
        """Check if hand contains at least one pair."""
        rank_counts = self.get_rank_counts()
        return any(count >= 2 for count in rank_counts.values())
    
    def has_full_house(self):
        """Check if hand has three of one rank and two of another."""
        rank_counts = self.get_rank_counts()
        values = sorted(rank_counts.values(), reverse=True)
        return len(values) >= 2 and values[0] >= 3 and values[1] >= 2

Debugging Explanation

Problem: The original Kangaroo class used a mutable default argument (contents=[]), which creates a single list instance shared across all Kangaroo objects that don't provide explicit contents.

Solution: Use None as the default and create a new list instance inside the method. This ensures each Kangaroo gets its own unique pouch contents.

class Calculator:
    """Calculator with comprehensive error handling."""
    
    def safe_divide(self, numerator, denominator):
        """Divide with exception handling for various error conditions."""
        try:
            result = numerator / denominator
            return result
            
        except ValueError:
            print("Error: Invalid value provided")
            return None
            
        except (TypeError, ZeroDivisionError) as e:
            if isinstance(e, TypeError):
                print("Error: Invalid operation types")
            else:
                print("Error: Division by zero is not allowed")
            return None
            
        except Exception as e:
            print(f"Unexpected error: {e}")
            return None
            
        finally:
            print("Division operation completed")
    
    def process_calculation(self, operation, *args):
        """Process various mathematical operations with error handling."""
        try:
            if operation == 'divide':
                return self.safe_divide(args[0], args[1])
            elif operation == 'power':
                return args[0] ** args[1]
            else:
                raise ValueError(f"Unknown operation: {operation}")
                
        except ValueError as e:
            print(f"Value error: {e}")
            return None
        except Exception as e:
            print(f"Calculation error: {e}")
            return None

# Demonstration

# Set operations for robot capability management
class RobotCapabilityManager:
    """Manages robot capabilities using set operations."""
    
    def __init__(self):
        self.available_capabilities = set()
        self.robot_skills = {}
    
    def analyze_team_capabilities(self, robot_team):
        """Use set operations to analyze team capabilities."""
        # Union: Combine all team capabilities
        team_capabilities = set()
        for robot in robot_team:
            team_capabilities = team_capabilities.union(self.robot_skills.get(robot, set()))
        
        # Intersection: Find common capabilities
        common_capabilities = set.intersection(*[self.robot_skills.get(robot, set()) for robot in robot_team])
        
        # Difference: Identify missing capabilities
        required_capabilities = {'navigation', 'object_manipulation', 'communication'}
        missing_capabilities = required_capabilities.difference(team_capabilities)
        
        # Symmetric Difference: Find unique capabilities
        if len(robot_team) == 2:
            robot1_skills = self.robot_skills.get(robot_team[0], set())
            robot2_skills = self.robot_skills.get(robot_team[1], set())
            unique_capabilities = robot1_skills.symmetric_difference(robot2_skills)
        
        return {
            'team_capabilities': team_capabilities,
            'common_capabilities': common_capabilities,
            'missing_capabilities': missing_capabilities,
            'unique_capabilities': unique_capabilities if len(robot_team) == 2 else set()
        }

# Example usage

}

2.2 Linear Search Implementation

def linear_search(arr, target):
    """
    Perform linear search on a list.
    
    Args:
        arr: List to search through
        target: Element to find
    
    Returns:
        Index of target if found, -1 otherwise
    """
    for i in range(len(arr)):
        if arr[i] == target:
            return i
    return -1

def linear_search_with_sentinel(arr, target):
    """
    Optimized linear search using sentinel technique.
    """
    # Add sentinel at the end
    arr.append(target)
    i = 0
    while arr[i] != target:
        i += 1
    # Remove sentinel
    arr.pop()
    return i if i < len(arr) else -1

# Robot application example
class RobotTaskScheduler:
    """Uses linear search for task scheduling."""
    
    def __init__(self):
        self.tasks = []
        self.completed_tasks = []
    
    def find_task_by_priority(self, priority):
        """Find task with specific priority using linear search."""
        return linear_search([task['priority'] for task in self.tasks], priority)
    
    def schedule_task(self, task_name, priority):
        """Add task to scheduler."""
        self.tasks.append({'name': task_name, 'priority': priority})
    
    def complete_highest_priority(self):
        """Complete highest priority task using linear search."""
        if not self.tasks:
            return None
        
        highest_priority = float('inf')
        task_index = -1
        
        for i, task in enumerate(self.tasks):
            if task['priority'] < highest_priority:
                highest_priority = task['priority']
                task_index = i
        
        if task_index != -1:
            completed_task = self.tasks.pop(task_index)
            self.completed_tasks.append(completed_task)
            return completed_task
        return None
