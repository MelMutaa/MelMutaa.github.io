Discussion: Data Structures in OOP Context

Data structures are fundamental to object-oriented development as they provide the mechanisms for organizing, storing, and manipulating object relationships and state. Here are three key data structures and their roles in supporting OOP principles:

1. Linked Lists

OOP Support: Encapsulation and relationship management

Use Case: Managing object sequences and hierarchies

class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None
    
    def add(self, data):
        new_node = Node(data)
        if not self.head:
            self.head = new_node
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = new_node

OOP Benefits: Each node encapsulates data and relationship, demonstrating composition.

2. Hash Maps / Dictionaries

OOP Support: Efficient object storage and retrieval

Use Case: Object registries, caching, configuration

class ObjectRegistry:
    def __init__(self):
        self._registry = {}
    
    def register(self, key, obj):
        self._registry[key] = obj
    
    def get(self, key):
        return self._registry.get(key)
    
    def get_all(self):
        return self._registry.items()

OOP Benefits: Enables object lookup by key, supporting polymorphism and factory patterns.

3. Trees

OOP Support: Hierarchical object relationships

Use Case: Organizational structures, file systems, UI components

class TreeNode:
    def __init__(self, value):
        self.value = value
        self.children = []
    
    def add_child(self, child_node):
        self.children.append(child_node)
    
    def traverse(self):
        results = [self.value]
        for child in self.children:
            results.extend(child.traverse())
        return results

OOP Benefits: Natural representation of inheritance and composition hierarchies.

Practical Integration Example

class EmployeeManagementSystem:
    """Demonstrates multiple data structures in OOP context."""
    
    def __init__(self):
        # Dictionary for quick employee lookup
        self.employee_registry = {}
        # Tree for organizational hierarchy
        self.department_tree = {}
        # Linked list for task scheduling
        self.task_queue = LinkedList()
    
    def add_employee(self, emp_id, name, department):
        """Add employee using dictionary for O(1) access."""
        employee = Employee(emp_id, name, department)
        self.employee_registry[emp_id] = employee
        
        # Maintain department hierarchy (tree structure)
        if department not in self.department_tree:
            self.department_tree[department] = []
        self.department_tree[department].append(employee)
    
    def schedule_task(self, task_description):
        """Use linked list for task sequencing."""
        self.task_queue.add(task_description)
    
    def get_department_employees(self, department):
        """Tree-based department query."""
        return self.department_tree.get(department, [])
    
    def get_next_task(self):
        """Linked list sequential access."""
        if self.task_queue.head:
            task = self.task_queue.head.data
            self.task_queue.head = self.task_queue.head.next
            return task
        return None

class Employee:
    def __init__(self, emp_id, name, department):
        self.emp_id = emp_id
        self.name = name
        self.department = department
    
    def __str__(self):
        return f"{self.name} ({self.emp_id}) - {self.department}"

Nested Dictionary Implementation

This activity demonstrates creating a Car class that uses nested dictionaries to store complex vehicle data and utilizes dictionary methods for data manipulation.

class Car:
    """Car class with nested dictionary for comprehensive vehicle data."""
    
    def __init__(self, make, model, year):
        self.make = make
        self.model = model
        self.year = year
        # Nested dictionary structure
        self.specifications = {
            'engine': {
                'type': '',
                'displacement': 0,
                'horsepower': 0,
                'fuel_type': ''
            },
            'dimensions': {
                'length': 0,
                'width': 0,
                'height': 0,
                'weight': 0
            },
            'performance': {
                'top_speed': 0,
                'acceleration_0_60': 0,
                'fuel_economy': {
                    'city': 0,
                    'highway': 0,
                    'combined': 0
                }
            },
            'features': []
        }
        self.maintenance_history = {}
    
    def set_engine_specs(self, engine_type, displacement, horsepower, fuel_type):
        """Update engine specifications in nested dictionary."""
        self.specifications['engine'] = {
            'type': engine_type,
            'displacement': displacement,
            'horsepower': horsepower,
            'fuel_type': fuel_type
        }
    
    def add_maintenance_record(self, date, service_type, cost, mileage):
        """Add maintenance record to history dictionary."""
        self.maintenance_history[date] = {
            'service_type': service_type,
            'cost': cost,
            'mileage': mileage
        }
    
    def demonstrate_dictionary_methods(self):
        """Demonstrate items(), keys(), values() methods."""
        
        print("=== Dictionary Methods Demonstration ===")
        
        # items() - returns key-value pairs
        print("\n1. items() method - Engine specifications:")
        for key, value in self.specifications['engine'].items():
            print(f"   {key}: {value}")
        
        # keys() - returns all keys
        print("\n2. keys() method - Specification categories:")
        categories = list(self.specifications.keys())
        print(f"   Available categories: {categories}")
        
        # values() - returns all values
        print("\n3. values() method - Engine specification values:")
        engine_values = list(self.specifications['engine'].values())
        print(f"   Engine values: {engine_values}")
        
        # Working with nested items()
        print("\n4. Nested items() - All specifications:")
        for category, details in self.specifications.items():
            print(f"   {category.upper()}:")
            if isinstance(details, dict):
                for key, value in details.items():
                    if isinstance(value, dict):
                        print(f"     {key}:")
                        for sub_key, sub_value in value.items():
                            print(f"       {sub_key}: {sub_value}")
                    else:
                        print(f"     {key}: {value}")
            else:
                print(f"     {details}")
    
    def find_specification(self, search_key):
        """Search for a specification across nested dictionaries."""
        results = []
        
        def search_nested_dict(d, key, path=""):
            if isinstance(d, dict):
                for k, v in d.items():
                    current_path = f"{path}.{k}" if path else k
                    if k == search_key:
                        results.append((current_path, v))
                    if isinstance(v, dict):
                        search_nested_dict(v, key, current_path)
                    elif isinstance(v, list):
                        for item in v:
                            if isinstance(item, dict):
                                search_nested_dict(item, key, current_path)
        
        search_nested_dict(self.specifications, search_key)
        return results
    
    def get_car_summary(self):
        """Generate a summary using dictionary methods."""
        summary = {
            'basic_info': f"{self.year} {self.make} {self.model}",
            'engine_info': self.specifications['engine'],
            'maintenance_records': len(self.maintenance_history),
            'recent_maintenance': list(self.maintenance_history.keys())[-3:] if self.maintenance_history else []
        }
        return summary
    
    def __str__(self):
        return f"{self.year} {self.make} {self.model}"

Dictionary Methods in Practice

items() - Iterating through key-value pairs

Perfect for processing all data in a dictionary, especially useful for nested structures and data transformation.

keys() - Accessing dictionary keys

Useful for validation, checking existence of required fields, and understanding the structure of your data.

values() - Working with dictionary values

Ideal for calculations, aggregations, and operations that don't require the keys but need all values.

This system implements a design pattern recommendation engine based on user requirements, inspired by the research in Zagorulko et al (2021). The program uses constructor-based initialization and processes user input to suggest appropriate design patterns. 

class DesignPatternRecommender:
    """
    Intelligent design pattern recommendation system.
    Based on Zagorulko et al (2021) research on pattern selection.
    """
    
    def __init__(self):
        # Initialize pattern database with constructor-based setup
        self.patterns = self._initialize_patterns()
        self.user_requirements = {}
        self.recommendation_history = []
    
    def _initialize_patterns(self):

        return {
            'creational': {
                'singleton': {
                    'name': 'Singleton',

                    'when_to_use': [

                    'complexity': 'low',

                'factory': {

                    'when_to_use': [

                    'flexibility': 'high'
                },
                'builder': {
                    'name': 'Builder',
                    'purpose': 'Construct complex objects step by step',
                    'when_to_use': [
                        'complex object construction',
                        'multiple construction representations',
                        'step-by-step object building'
                    ],
                    'complexity': 'medium',
                    'flexibility': 'high'
                }
            },
            'structural': {
                'adapter': {
                    'name': 'Adapter',
                    'purpose': 'Make incompatible interfaces work together',
                    'when_to_use': [
                        'integrating legacy systems',
                        'interface compatibility needed',
                        'wrapper for third-party libraries'
                    ],
                    'complexity': 'low',
                    'flexibility': 'medium'
                },
                'composite': {
                    'name': 'Composite',
                    'purpose': 'Treat individual objects and compositions uniformly',
                    'when_to_use': [
                        'tree-like object structures',

                }
            },
            'behavioral': {
                'observer': {

                    'purpose': 'Define one-to-many dependency between objects',

                    ],
                    'complexity': 'medium',
                    'flexibility': 'high'
                }
            }
        }
    
    def collect_user_requirements(self):
        """Collect and process user requirements using constructor-like input."""
        print("=== Design Pattern Recommendation System ===")
        print("Based on Zagorulko et al (2021) research\n")
        
        # Initialize requirements through user input
        self.user_requirements = {
            'problem_domain': input("What problem domain are you working in? (e.g., UI, Database, Networking): ").strip().lower(),
            'primary_concern': input("What is your primary concern? (e.g., object creation, structure, behavior): ").strip().lower(),
            'complexity_tolerance': input("Complexity tolerance? (low/medium/high): ").strip().lower(),
            'flexibility_need': input("Flexibility requirement? (low/medium/high): ").strip().lower(),
            'specific_needs': input("Describe specific needs (comma-separated): ").strip().lower().split(',')
        }
        
        # Process and clean input
        self.user_requirements['specific_needs'] = [
            need.strip() for need in self.user_requirements['specific_needs'] 
            if need.strip()
        ]
    
    def calculate_pattern_score(self, pattern, requirements):
        """Calculate matching score between pattern and requirements."""
        score = 0
        
        # Match based on primary concern
        concern_mapping = {
            'object creation': 'creational',
            'structure': 'structural', 
            'behavior': 'behavioral'
        }
        
        primary_concern = requirements['primary_concern']
        if primary_concern in concern_mapping:
            pattern_category = next(
                (cat for cat, patterns in self.patterns.items() 
                 if pattern in patterns), 
                None
            )
            if pattern_category == concern_mapping[primary_concern]:
                score += 3
        
        # Match based on complexity
        if pattern['complexity'] == requirements['complexity_tolerance']:
            score += 2
        elif self._complexity_distance(pattern['complexity'], requirements['complexity_tolerance']) == 1:
            score += 1
        
        # Match based on flexibility
        if pattern['flexibility'] == requirements['flexibility_need']:
            score += 2
        elif self._flexibility_match(pattern['flexibility'], requirements['flexibility_need']):
            score += 1
        
        # Match based on specific needs keywords
        for need in requirements['specific_needs']:
            for use_case in pattern['when_to_use']:
                if need in use_case:
                    score += 2
                    break
        
        return score
    
    def _complexity_distance(self, pattern_complexity, user_complexity):
        """Calculate distance between complexity levels."""
        levels = ['low', 'medium', 'high']
        try:
            return abs(levels.index(pattern_complexity) - levels.index(user_complexity))
        except ValueError:
            return 2  # Maximum distance
    
    def _flexibility_match(self, pattern_flex, user_flex):
        """Check if flexibility requirements are compatible."""
        # High user flexibility need can accept medium/high pattern flexibility
        # Low user flexibility need prefers low pattern flexibility
        flex_levels = {'low': 0, 'medium': 1, 'high': 2}
        try:
            return flex_levels[pattern_flex] >= flex_levels[user_flex]
        except KeyError:
            return False
    
    def generate_recommendations(self):
        """Generate pattern recommendations based on user input."""
        recommendations = []
        
        for category, patterns in self.patterns.items():
            for pattern_key, pattern_data in patterns.items():
                score = self.calculate_pattern_score(pattern_data, self.user_requirements)
                if score > 0:
                    recommendations.append({
                        'pattern': pattern_data['name'],
                        'category': category,
                        'score': score,
                        'purpose': pattern_data['purpose'],
                        'when_to_use': pattern_data['when_to_use']
                    })
        
        # Sort by score descending
        recommendations.sort(key=lambda x: x['score'], reverse=True)
        return recommendations[:5]  # Return top 5 recommendations
    
    def display_recommendations(self, recommendations):
        """Display recommendations to user."""
        print(f"\n=== RECOMMENDATIONS FOR YOUR PROJECT ===")
        print(f"Problem Domain: {self.user_requirements['problem_domain']}")
        print(f"Primary Concern: {self.user_requirements['primary_concern']}")
        print(f"Specific Needs: {', '.join(self.user_requirements['specific_needs'])}")
        print("\nTop Pattern Recommendations:\n")
        
        for i, rec in enumerate(recommendations, 1):
            print(f"{i}. {rec['pattern']} ({rec['category'].title()})")
            print(f"   Match Score: {rec['score']}/10")
            print(f"   Purpose: {rec['purpose']}")
            print(f"   Best for: {', '.join(rec['when_to_use'][:2])}...")
            print()
    
    def save_recommendation_session(self):
        """Save current session to history."""
        session_data = {
            'timestamp': '2024-01-15 10:30:00',  # In real implementation, use datetime.now()
            'requirements': self.user_requirements,
            'recommendations': self.generate_recommendations()[:3]
        }
        self.recommendation_history.append(session_data)
    
    def run(self):
        """Main method to run the recommendation system."""
        self.collect_user_requirements()
        recommendations = self.generate_recommendations()
        self.display_recommendations(recommendations)
        self.save_recommendation_session()
        
        # Show connection to Zagorulko et al research
        print("=== RESEARCH CONNECTION ===")
