"""
Factorial calculation using recursion.
This module demonstrates recursive factorial calculation with proper error handling.
"""

def factorial(x):
    """
    Calculate the factorial of a non-negative integer using recursion.
    
    Args:
        x (int): Non-negative integer for which to calculate factorial
        
    Returns:
        int: Factorial of x
        
    Raises:
        ValueError: If x is negative
        TypeError: If x is not an integer
    """
    # Input validation
    if not isinstance(x, int):
        raise TypeError("Input must be an integer")
    if x < 0:
        raise ValueError("Factorial is not defined for negative numbers")
    
    # Base case
    if x == 0 or x == 1:
        return 1
    
    # Recursive case
    return x * factorial(x - 1)


if __name__ == "__main__":
    num = 3
    print("The factorial of", num, "is", factorial(num))

"""
Simple Calculator Application
A command-line calculator that performs basic arithmetic operations.
This module demonstrates good documentation practices and user input handling.
"""

def add(x, y):
    """
    Add two numbers and return the result.
    
    Args:
        x (float): First number to add
        y (float): Second number to add
        
    Returns:
        float: Sum of x and y
    """
    return x + y


def subtract(x, y):
    """
    Subtract the second number from the first.
    
    Args:
        x (float): Number to subtract from (minuend)
        y (float): Number to subtract (subtrahend)
        
    Returns:
        float: Result of x - y
    """
    return x - y


def multiply(x, y):
    """
    Multiply two numbers.
    
    Args:
        x (float): First factor
        y (float): Second factor
        
    Returns:
        float: Product of x and y
    """
    return x * y


def divide(x, y):
    """
    Divide the first number by the second.
    
    Args:
        x (float): Dividend (number to be divided)
        y (float): Divisor (number to divide by)
        
    Returns:
        float: Result of x / y
        
    Raises:
        ZeroDivisionError: If y is zero
    """
    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero")
    return x / y


def display_menu():
    """
    Display the calculator operation menu to the user.
    """
    print("\n" + "="*40)
    print("           SIMPLE CALCULATOR")
    print("="*40)
    print("Select operation:")
    print("1. Add")
    print("2. Subtract") 
    print("3. Multiply")
    print("4. Divide")
    print("="*40)


def get_user_choice():
    """
    Get and validate user's operation choice.
    
    Returns:
        str: Validated user choice ('1', '2', '3', or '4')
    """
    while True:
        choice = input("Enter choice (1/2/3/4): ").strip()
        if choice in ('1', '2', '3', '4'):
            return choice
        print("Invalid input! Please enter 1, 2, 3, or 4.")


def get_numbers():
    """
    Get two numbers from the user with validation.
    
    Returns:
        tuple: Two valid floating-point numbers (num1, num2)
    """
    while True:
        try:
            num1 = float(input("Enter first number: "))
            num2 = float(input("Enter second number: "))
            return num1, num2
        except ValueError:
            print("Invalid number! Please enter numeric values.")


def perform_calculation(choice, num1, num2):
    """
    Perform the selected calculation operation.
    
    Args:
        choice (str): User's operation choice
        num1 (float): First number
        num2 (float): Second number
        
    Returns:
        float: Result of the calculation
    """
    # Dictionary mapping choices to operation functions
    operations = {
        '1': add,
        '2': subtract, 
        '3': multiply,
        '4': divide
    }
    
    operation_func = operations[choice]
    return operation_func(num1, num2)


def main():
    """
    Main calculator program loop.
    Handles user interaction and calculation flow.
    """
    print("Welcome to the Simple Calculator!")
    
    while True:
        # Display menu and get user input
        display_menu()
        choice = get_user_choice()
        num1, num2 = get_numbers()
        
        try:
            # Perform calculation and display result
            result = perform_calculation(choice, num1, num2)
            operation_symbols = {'1': '+', '2': '-', '3': '*', '4': '/'}
            symbol = operation_symbols[choice]
            
            print(f"\nResult: {num1} {symbol} {num2} = {result}")
            
        except ZeroDivisionError:
            print("\nError: Cannot divide by zero!")
        except Exception as e:
            print(f"\nAn error occurred: {e}")
        
        # Ask if user wants to continue
        while True:
            continue_calc = input("\nPerform another calculation? (yes/no): ").strip().lower()
            if continue_calc in ('yes', 'y', 'no', 'n'):
                break
            print("Please enter 'yes' or 'no'")
        
        if continue_calc in ('no', 'n'):
            print("\nThank you for using the calculator. Goodbye!")
            break


if __name__ == "__main__":
    main()

Key Findings from Rani et al. (2021)

Rani et al. (2021) conducted empirical research on code commenting practices and their impact on software maintenance. The key findings that influenced our documentation approach:

1. Intent-Oriented Comments

Research Finding: Comments explaining "why" rather than "what" are more valuable for maintenance.

Our Application: Docstrings focus on purpose and reasoning behind operations.

2. Consistent Documentation Standards

Research Finding: Standardized formats improve comprehension and reduce cognitive load.

Our Application: Consistent use of Google-style docstrings with Args/Returns sections.

3. Maintenance-Focused Documentation

Research Finding: Documentation should anticipate future changes and maintenance needs.

Our Application: Clear error conditions and modification guidelines in docstrings.

Specific Research-Based Improvements

  • Parameter Validation Documentation: Rani et al. emphasize documenting input constraints and validation logic
  • Exception Documentation: Research shows that documenting potential exceptions reduces debugging time
  • Usage Examples: The study found that examples in documentation significantly improve understanding
  • Maintenance Notes: Documenting known limitations and potential improvements aids future developers

def divide(x, y):
    """
    Divide the first number by the second.
    
    Based on Rani et al. (2021) recommendations:
    - Clearly states purpose and mathematical operation
    - Documents all parameters with types and roles
    - Explicitly mentions exception conditions
    - Provides maintenance guidance for edge cases
    
    Args:
        x (float): Dividend (number to be divided)
        y (float): Divisor (number to divide by)
        
    Returns:
        float: Result of x / y
        
    Raises:
        ZeroDivisionError: If y is zero - documented per research findings
        TypeError: If inputs are not numeric - anticipating maintenance needs
        
    Example:
        >>> divide(10, 2)
        5.0
        >>> divide(5, 0)
        ZeroDivisionError: Cannot divide by zero
        
    Note:
        For financial calculations, consider using decimal.Decimal
        instead of float to avoid floating-point precision issues.
    """
    if not isinstance(x, (int, float)) or not isinstance(y, (int, float)):
        raise TypeError("Both inputs must be numeric")
    if y == 0:
        raise ZeroDivisionError("Cannot divide by zero")
    return x / y

"""
Unit tests for Simple Calculator Application
Comprehensive test suite covering all calculator operations and edge cases.
"""

import unittest
import sys
from io import StringIO
from unittest.mock import patch


class TestCalculatorOperations(unittest.TestCase):
    """Test cases for individual calculator operations."""
    
    def test_add_positive_numbers(self):
        """Test addition with positive numbers."""
        from calculator import add
        self.assertEqual(add(5, 3), 8)
        self.assertEqual(add(2.5, 3.7), 6.2)
    
    def test_add_negative_numbers(self):
        """Test addition with negative numbers."""
        from calculator import add
        self.assertEqual(add(-5, -3), -8)
        self.assertEqual(add(-2.5, 3.5), 1.0)
    
    def test_add_zero(self):
        """Test addition with zero."""
        from calculator import add
        self.assertEqual(add(0, 5), 5)
        self.assertEqual(add(5, 0), 5)
        self.assertEqual(add(0, 0), 0)
    
    def test_subtract_positive_numbers(self):
        """Test subtraction with positive numbers."""
        from calculator import subtract
        self.assertEqual(subtract(10, 4), 6)
        self.assertEqual(subtract(5.5, 2.2), 3.3)
    
    def test_subtract_negative_numbers(self):
        """Test subtraction with negative numbers."""
        from calculator import subtract
        self.assertEqual(subtract(-5, -3), -2)
        self.assertEqual(subtract(5, -3), 8)
    
    def test_multiply_positive_numbers(self):
        """Test multiplication with positive numbers."""
        from calculator import multiply
        self.assertEqual(multiply(4, 3), 12)
        self.assertEqual(multiply(2.5, 4), 10.0)
    
    def test_multiply_by_zero(self):
        """Test multiplication by zero."""
        from calculator import multiply
        self.assertEqual(multiply(5, 0), 0)
        self.assertEqual(multiply(0, 5), 0)
        self.assertEqual(multiply(0, 0), 0)
    
    def test_divide_positive_numbers(self):
        """Test division with positive numbers."""
        from calculator import divide
        self.assertEqual(divide(10, 2), 5.0)
        self.assertEqual(divide(5, 2), 2.5)
    
    def test_divide_by_zero(self):
        """Test division by zero raises appropriate error."""
        from calculator import divide
        with self.assertRaises(ZeroDivisionError):
            divide(10, 0)
        with self.assertRaises(ZeroDivisionError):
            divide(0, 0)
    
    def test_divide_negative_numbers(self):
        """Test division with negative numbers."""
        from calculator import divide
        self.assertEqual(divide(-10, 2), -5.0)
        self.assertEqual(divide(10, -2), -5.0)
        self.assertEqual(divide(-10, -2), 5.0)

class TestUserInterface(unittest.TestCase):
    """Tests for user interface components and input handling."""
    
    def test_display_menu_output(self):
        """Test that the menu displays correctly."""
        from calculator import display_menu
        with patch('sys.stdout', new_callable=StringIO) as mock_stdout:
            display_menu()
            output = mock_stdout.getvalue()
            self.assertIn("SIMPLE CALCULATOR", output)
            self.assertIn("1. Add", output)
            self.assertIn("4. Divide", output)
    
    def test_get_user_choice_valid_input(self):
        """Test valid user input handling."""
        from calculator import get_user_choice
        with patch('builtins.input', return_value='2'):
            choice = get_user_choice()
            self.assertEqual(choice, '2')
    
    def test_get_user_choice_invalid_then_valid(self):
        """Test recovery from invalid input."""
        from calculator import get_user_choice
        with patch('builtins.input', side_effect=['5', '3']):
            with patch('sys.stdout', new_callable=StringIO) as mock_stdout:
                choice = get_user_choice()
                output = mock_stdout.getvalue()
                self.assertIn("Invalid input", output)
                self.assertEqual(choice, '3')
    
    def test_get_numbers_valid_input(self):
        """Test valid number input."""
        from calculator import get_numbers
        with patch('builtins.input', side_effect=['10', '5']):
            num1, num2 = get_numbers()
            self.assertEqual(num1, 10.0)
            self.assertEqual(num2, 5.0)
    
    def test_get_numbers_invalid_then_valid(self):
        """Test recovery from invalid number input."""
        from calculator import get_numbers
        with patch('builtins.input', side_effect=['abc', '10', '5']):
            with patch('sys.stdout', new_callable=StringIO) as mock_stdout:
                num1, num2 = get_numbers()
                output = mock_stdout.getvalue()
                self.assertIn("Invalid number", output)
                self.assertEqual(num1, 10.0)
                self.assertEqual(num2, 5.0)


class TestEdgeCases(unittest.TestCase):
    """Tests for edge cases and boundary conditions."""
    
    def test_large_numbers(self):
        """Test operations with very large numbers."""
        from calculator import add, multiply
        large_num = 1e308
        self.assertEqual(add(large_num, large_num), large_num * 2)
        # Note: Multiplication might cause overflow in some cases
    
    def test_small_numbers(self):
        """Test operations with very small numbers."""
        from calculator import multiply, divide
        small_num = 1e-308
        self.assertEqual(multiply(small_num, 2), small_num * 2)
        self.assertEqual(divide(small_num, 2), small_num / 2)
    
    def test_floating_point_precision(self):
        """Test floating point precision handling."""
        from calculator import add, subtract
        # Test that floating point arithmetic works as expected
        self.assertAlmostEqual(add(0.1, 0.2), 0.3, places=7)
        self.assertAlmostEqual(subtract(0.3, 0.1), 0.2, places=7)


def run_comprehensive_tests():
    """
    Run all test suites and provide detailed reporting.
    """
    # Create test suite
    loader = unittest.TestLoader()
    suite = unittest.TestSuite()
    
    # Add all test cases
    suite.addTests(loader.loadTestsFromTestCase(TestCalculatorOperations))
    suite.addTests(loader.loadTestsFromTestCase(TestCalculatorIntegration))
    suite.addTests(loader.loadTestsFromTestCase(TestUserInterface))
    suite.addTests(loader.loadTestsFromTestCase(TestEdgeCases))
    
    # Run tests with detailed reporting
    runner = unittest.TextTestRunner(verbosity=2)
    result = runner.run(suite)
    
    # Print summary
    print(f"\n{'='*50}")
    print("TEST SUMMARY")
    print(f"{'='*50}")
    print(f"Tests run: {result.testsRun}")
    print(f"Failures: {len(result.failures)}")
    print(f"Errors: {len(result.errors)}")
    print(f"Success rate: {(result.testsRun - len(result.failures) - len(result.errors)) / result.testsRun * 100:.1f}%")
    
    return result.wasSuccessful()


if __name__ == "__main__":
    # Run the comprehensive test suite
    success = run_comprehensive_tests()
    
    # Exit with appropriate code
    sys.exit(0 if success else 1)
