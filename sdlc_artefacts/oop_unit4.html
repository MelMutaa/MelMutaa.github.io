Task 3 — Python Implementation

Employee Management System Code

Complete Python Implementation

This Python program implements the UML class diagram from Task 2, providing full employee management and leave booking functionality.

from datetime import date, datetime
from typing import List, Optional
import uuid

class Employee:
    """Represents an employee in the organization."""
    
    def __init__(self, employee_id: str, name: str, email: str, phone: str, 
                 position: str, salary: float, hire_date: date, department: str = "General"):
        self.employee_id = employee_id
        self.name = name
        self.email = email
        self.phone = phone
        self.position = position
        self.salary = salary
        self.hire_date = hire_date
        self.department = department
        self.leave_balance = 25  # Default annual leave days
        self.leave_requests = []
    
    def apply_leave(self, start_date: date, end_date: date, leave_type: str = "Annual", reason: str = "") -> str:
        """Apply for leave and return request ID."""
        if self.leave_balance <= 0 and leave_type == "Annual":
            raise ValueError("Insufficient leave balance")
        
        duration = (end_date - start_date).days + 1
        if leave_type == "Annual" and duration > self.leave_balance:
            raise ValueError(f"Requested {duration} days but only {self.leave_balance} available")
        
        request_id = str(uuid.uuid4())[:8]
        leave_request = LeaveRequest(request_id, self.employee_id, start_date, end_date, leave_type, reason)
        self.leave_requests.append(leave_request)
        
        return request_id
    
    def update_info(self, email: Optional[str] = None, phone: Optional[str] = None, 
                   position: Optional[str] = None) -> None:
        """Update employee contact information or position."""
        if email:
            self.email = email
        if phone:
            self.phone = phone
        if position:
            self.position = position
    
    def get_leave_balance(self) -> int:
        """Get current annual leave balance."""
        return self.leave_balance
    
    def calculate_service_years(self) -> float:
        """Calculate years of service."""
        today = date.today()
        service_days = (today - self.hire_date).days
        return service_days / 365.25
    
    def __str__(self) -> str:
        return f"Employee {self.employee_id}: {self.name} ({self.position})"

class LeaveRequest:
    """Represents a leave request made by an employee."""
    
    def __init__(self, request_id: str, employee_id: str, start_date: date, 
                 end_date: date, leave_type: str, reason: str = ""):
        self.request_id = request_id
        self.employee_id = employee_id
        self.start_date = start_date
        self.end_date = end_date
        self.leave_type = leave_type
        self.reason = reason
        self.status = "Pending"  # Pending, Approved, Rejected, Cancelled
        self.approved_date = None
        self.approved_by = None
    
    def approve(self, approved_by: str) -> None:
        """Approve the leave request."""
        self.status = "Approved"
        self.approved_date = datetime.now()
        self.approved_by = approved_by
    
    def reject(self, rejected_by: str) -> None:
        """Reject the leave request."""
        self.status = "Rejected"
        self.approved_date = datetime.now()
        self.approved_by = rejected_by
    
    def cancel(self) -> None:
        """Cancel the leave request."""
        if self.status == "Pending":
            self.status = "Cancelled"
        else:
            raise ValueError("Only pending requests can be cancelled")
    
    def get_duration(self) -> int:
        """Calculate the duration of leave in days."""
        return (self.end_date - self.start_date).days + 1
    
    def __str__(self) -> str:
        return f"LeaveRequest {self.request_id}: {self.employee_id} ({self.status})"

class Department:
    """Represents a department in the organization."""
    
    def __init__(self, dept_id: str, name: str, manager: str, budget: float = 0.0):
        self.dept_id = dept_id
        self.name = name
        self.manager = manager
        self.budget = budget
        self.employees = []
    
    def add_employee(self, employee: Employee) -> None:
        """Add an employee to the department."""
        employee.department = self.name
        self.employees.append(employee)
    
    def remove_employee(self, employee_id: str) -> bool:
        """Remove an employee from the department."""
        for i, emp in enumerate(self.employees):
            if emp.employee_id == employee_id:
                self.employees.pop(i)
                emp.department = "General"
                return True
        return False
    
    def get_employee_count(self) -> int:
        """Get the number of employees in the department."""
        return len(self.employees)
    
    def get_total_salary_budget(self) -> float:
        """Calculate total salary budget for the department."""
        return sum(emp.salary for emp in self.employees)
    
    def __str__(self) -> str:
        return f"Department {self.dept_id}: {self.name}"

class LeaveManagementSystem:
    """Main system for managing employees and leave requests."""
    
    def __init__(self):
        self.employees = {}
        self.departments = {}
        self.leave_requests = {}
    
    def add_employee(self, employee: Employee) -> None:
        """Add an employee to the system."""
        self.employees[employee.employee_id] = employee
    
    def get_employee(self, employee_id: str) -> Optional[Employee]:
        """Retrieve an employee by ID."""
        return self.employees.get(employee_id)
    
    def process_leave_request(self, request_id: str, action: str, processed_by: str) -> bool:
        """Process a leave request (approve or reject)."""
        # Find the request in all employees' leave requests
        for employee in self.employees.values():
            for request in employee.leave_requests:
                if request.request_id == request_id:
                    if action.lower() == "approve":
                        request.approve(processed_by)
                        if request.leave_type == "Annual":
                            employee.leave_balance -= request.get_duration()
                        return True
                    elif action.lower() == "reject":
                        request.reject(processed_by)
                        return True
        return False
    
    def get_pending_requests(self) -> List[LeaveRequest]:
        """Get all pending leave requests."""
        pending = []
        for employee in self.employees.values():
            for request in employee.leave_requests:
                if request.status == "Pending":
                    pending.append(request)
        return pending

# Example usage and demonstration
def demonstrate_system():
    """Demonstrate the employee management system functionality."""
    print("=== Employee Management System Demonstration ===\n")
    
    # Create system instance
    lms = LeaveManagementSystem()
    
    # Create departments
    hr_dept = Department("D001", "Human Resources", "Alice Johnson", 500000)
    it_dept = Department("D002", "IT Department", "Bob Smith", 750000)
    
    # Create employees
    emp1 = Employee("E001", "John Doe", "john.doe@company.com", "555-0101", 
                   "HR Specialist", 50000, date(2020, 3, 15))
    emp2 = Employee("E002", "Jane Smith", "jane.smith@company.com", "555-0102",
                   "Software Developer", 70000, date(2019, 7, 22))
    
    # Add employees to system and departments
    lms.add_employee(emp1)
    lms.add_employee(emp2)
    hr_dept.add_employee(emp1)
    it_dept.add_employee(emp2)
    
    print("Employees created:")
    print(f"1. {emp1}")
    print(f"2. {emp2}")
    print(f"   Department: {emp1.department}")
    print(f"   Leave Balance: {emp1.get_leave_balance()} days\n")
    
    # Demonstrate leave application
    try:
        request_id = emp1.apply_leave(
            start_date=date(2024, 6, 10),
            end_date=date(2024, 6, 14),
            leave_type="Annual",
            reason="Summer vacation"
        )
        print(f"Leave request submitted: {request_id}")
        
        # Process the leave request
        lms.process_leave_request(request_id, "approve", "HR Manager")
        print("Leave request approved by HR Manager")
        
        # Check updated balance
        print(f"Updated leave balance: {emp1.get_leave_balance()} days\n")
        
    except ValueError as e:
        print(f"Error: {e}")
    
    # Show pending requests
    pending = lms.get_pending_requests()
    print(f"Pending leave requests: {len(pending)}")
    
    # Department information
    print(f"\nDepartment Information:")
    print(f"HR Department: {hr_dept.get_employee_count()} employees")
    print(f"IT Department: {it_dept.get_employee_count()} employees")

if __name__ == "__main__":
    demonstrate_system()
