Basic Encapsulation Example

class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.__balance = balance  # Private attribute

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
            return f"Deposited £{amount}"
        return "Invalid deposit amount"

    def withdraw(self, amount):
        if 0 < amount <= self.__balance:
            self.__balance -= amount
            return f"Withdrew £{amount}"
        return "Insufficient funds or invalid amount"

    def get_balance(self):
        return f"Balance: £{self.__balance}"

    # Property for controlled access
    @property
    def balance(self):
        return self.__balance

# Demonstration
account = BankAccount("Alice", 100)
print(account.deposit(50))    # Deposited £50
print(account.withdraw(20))   # Withdrew £20
print(account.get_balance())  # Balance: £130

Advanced Abstract Class Implementation

from abc import ABC, abstractmethod
from datetime import datetime

class AbstractBankAccount(ABC):
    """Abstract base class for all bank accounts."""
    
    def __init__(self, account_holder, initial_balance=0):
        self.account_holder = account_holder
        self._balance = initial_balance  # Protected attribute
        self._transaction_history = []
        self._account_number = self._generate_account_number()
    
    def _generate_account_number(self):
        """Generate a unique account number."""
        return f"ACC{datetime.now().strftime('%Y%m%d%H%M%S')}"
    
    @abstractmethod
    def withdraw(self, amount):
        """Withdraw money from account - to be implemented by subclasses."""
        pass
    
    @abstractmethod
    def get_account_type(self):
        """Return account type - to be implemented by subclasses."""
        pass
    
    def deposit(self, amount):
        """Concrete method for deposits."""
        if amount > 0:
            self._balance += amount
            self._transaction_history.append(
                f"{datetime.now()}: Deposited £{amount}"
            )
            return True
        return False
    
    def get_balance(self):
        """Get current balance."""
        return self._balance
    
    def get_transaction_history(self):
        """Get transaction history."""
        return self._transaction_history.copy()  # Return copy for encapsulation
    
    def __str__(self):
        return f"{self.get_account_type()} - {self.account_holder}"

class SavingsAccount(AbstractBankAccount):
    """Savings account implementation."""
    
    def __init__(self, account_holder, initial_balance=0, interest_rate=0.02):
        super().__init__(account_holder, initial_balance)
        self._interest_rate = interest_rate
    
    def withdraw(self, amount):
        if 0 < amount <= self._balance:
            self._balance -= amount
            self._transaction_history.append(
                f"{datetime.now()}: Withdrew £{amount}"
            )
            return True
        return False
    
    def get_account_type(self):
        return "Savings Account"
    
    def apply_interest(self):
        """Apply annual interest to savings."""
        interest = self._balance * self._interest_rate
        self._balance += interest
        self._transaction_history.append(
            f"{datetime.now()}: Applied interest £{interest:.2f}"
        )
        return interest

class CurrentAccount(AbstractBankAccount):
    """Current account with overdraft facility."""
    
    def __init__(self, account_holder, initial_balance=0, overdraft_limit=100):
        super().__init__(account_holder, initial_balance)
        self._overdraft_limit = overdraft_limit
    
    def withdraw(self, amount):
        available_balance = self._balance + self._overdraft_limit
        if 0 < amount <= available_balance:
            self._balance -= amount
            self._transaction_history.append(
                f"{datetime.now()}: Withdrew £{amount}"
            )
            return True
        return False
    
    def get_account_type(self):
        return "Current Account"

# Demonstration
savings = SavingsAccount("Bob", 1000, 0.03)
current = CurrentAccount("Charlie", 500, 200)

accounts = [savings, current]
for account in accounts:
    print(f"\n{account}")
    account.deposit(100)
    account.withdraw(50)
    print(f"Balance: £{account.get_balance()}")
    print(f"Transactions: {account.get_transaction_history()[-2:]}")

Key Learning Points

  • Encapsulation: Private attributes (__balance) protect data integrity
  • Abstraction: Abstract base class defines interface without full implementation
  • Polymorphism: Different account types implement withdraw differently
  • Information Hiding: Internal implementation details are hidden from users

OOP Quality Metrics and Design Principles

Evaluating software quality through object-oriented metrics

Essential OOP Metrics

These metrics help assess the quality and maintainability of object-oriented programs:

Cohesion

Definition: How closely related the responsibilities of a class are

High Cohesion: Class has a single, well-defined purpose

Low Cohesion: Class handles multiple unrelated tasks (God object)

Coupling

Definition: Degree of interdependence between classes

Low Coupling: Classes are independent and interact through interfaces

High Coupling: Changes in one class affect many others

Encapsulation

Definition: Bundling data with methods that operate on that data

Good Encapsulation: Private attributes with public accessors

Poor Encapsulation: Public attributes allowing direct modification

Inheritance Depth

Definition: Number of levels in inheritance hierarchy

Optimal: 2-4 levels for maintainability

Problematic: Very deep hierarchies become fragile

Applying Metrics to Banking Example

Metric Application in Banking System Quality Assessment
Cohesion SavingsAccount handles only savings-related operations High - Single responsibility
Coupling Accounts depend only on abstract base class Low - Interface-based interaction
Encapsulation Balance and transaction history are protected Good - Controlled access
Inheritance Depth Two-level hierarchy: Abstract → Concrete Optimal - Maintainable

Cohesion in Practice

A class should have a single responsibility. High cohesion makes code easier to understand, test, and maintain.

Coupling Considerations

Low coupling reduces the ripple effect of changes. Use interfaces and dependency injection to minimize coupling.

Encapsulation Benefits

Proper encapsulation prevents invalid state changes and makes APIs more stable during refactoring.

Complete Banking System Implementation

from abc import ABC, abstractmethod
from datetime import datetime
from typing import List

class BankOperation(ABC):
    """Interface for all bank operations."""
    
    @abstractmethod
    def execute(self) -> bool:
        """Execute the banking operation."""
        pass
    
    @abstractmethod
    def get_description(self) -> str:
        """Get human-readable description."""
        pass

class DepositOperation(BankOperation):
    """Concrete operation for deposits."""
    
    def __init__(self, account, amount):
        self.account = account
        self.amount = amount
    
    def execute(self) -> bool:
        return self.account.deposit(self.amount)
    
    def get_description(self) -> str:
        return f"Deposit £{self.amount} to {self.account.account_holder}"

class WithdrawOperation(BankOperation):
    """Concrete operation for withdrawals."""
    
    def __init__(self, account, amount):
        self.account = account
        self.amount = amount
    
    def execute(self) -> bool:
        return self.account.withdraw(self.amount)
    
    def get_description(self) -> str:
        return f"Withdraw £{self.amount} from {self.account.account_holder}"

class BankingSystem:
    """Main banking system using abstraction and encapsulation."""
    
    def __init__(self):
        self._accounts = {}
        self._operations = []
    
    def add_account(self, account):
        """Add account to system with encapsulation."""
        self._accounts[account._account_number] = account
    
    def process_operation(self, operation: BankOperation) -> bool:
        """Process any bank operation polymorphically."""
        result = operation.execute()
        if result:
            self._operations.append({
                'timestamp': datetime.now(),
                'operation': operation.get_description(),
                'success': True
            })
        return result
    
    def get_operation_history(self):
        """Get operation history with encapsulation."""
        return self._operations.copy()

# Demonstration
system = BankingSystem()
savings = SavingsAccount("David", 2000)
system.add_account(savings)

# Polymorphic operation processing
operations = [
    DepositOperation(savings, 500),
    WithdrawOperation(savings, 200),
    DepositOperation(savings, 1000)
]

print("Processing banking operations:")
for op in operations:
    success = system.process_operation(op)
    status = "✓" if success else "✗"
    print(f"{status} {op.get_description()}")

print(f"\nFinal balance: £{savings.get_balance()}")
print(f"Operation history: {system.get_operation_history()}")

Article Discussion: Agent-Based vs Component-Based Modelling

Aspect Agent-Based Modelling (ABM) Component-Based Modelling
Focus Autonomous agents with behaviors and interactions Software components with defined interfaces
Key Concept Emergent behavior from individual actions Structural decomposition and reuse
Application in Yu & Hou (2023) Simulates distributed intelligent agents collaborating Not directly addressed in the paper
Strengths Captures complex system dynamics and adaptation Excellent for software architecture and maintenance
Relation to OOP Agents as objects with autonomy and decision-making Components as encapsulated, reusable classes

Key Insights from Yu & Hou (2023):

  • Agent-based modelling enables simulation of complex adaptive systems
  • Intelligent agents can exhibit emergent behaviors not programmed explicitly
  • ABM provides better scalability for distributed systems than rigid component architectures
  • The approach aligns well with object-oriented principles of encapsulation and messaging
