Basic Encapsulation Example

class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.__balance = balance  # Private attribute

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount
            return f"Deposited £{amount}"
        return "Invalid deposit amount"

    def withdraw(self, amount):
        if 0 < amount <= self.__balance:
            self.__balance -= amount
            return f"Withdrew £{amount}"
        return "Insufficient funds or invalid amount"

    def get_balance(self):
        return f"Balance: £{self.__balance}"

# Demonstration
account = BankAccount("Alice", 100)
print(account.deposit(50))
print(account.withdraw(20))
print(account.get_balance())

Advanced Abstract Class Implementation

from abc import ABC, abstractmethod
from datetime import datetime

class AbstractBankAccount(ABC):
    """Abstract base class for all bank accounts."""
    
    def __init__(self, account_holder, initial_balance=0):
        self.account_holder = account_holder
        self._balance = initial_balance
        self._transaction_history = []
        self._account_number = self._generate_account_number()
    
    def _generate_account_number(self):
        return f"ACC{datetime.now().strftime('%Y%m%d%H%M%S')}"
    
    @abstractmethod
    def withdraw(self, amount):
        pass
    
    @abstractmethod
    def get_account_type(self):
        pass
    
    def deposit(self, amount):
        if amount > 0:
            self._balance += amount
            self._transaction_history.append(
                f"Deposited £{amount}"
            )
            return True
        return False
    
    def get_balance(self):
        return self._balance

class SavingsAccount(AbstractBankAccount):
    def __init__(self, account_holder, initial_balance=0):
        super().__init__(account_holder, initial_balance)
    
    def withdraw(self, amount):
        if 0 < amount <= self._balance:
            self._balance -= amount
            return True
        return False
    
    def get_account_type(self):
        return "Savings Account"

Key Learning Points

  • Encapsulation: Private attributes (__balance) protect data integrity
  • Abstraction: Abstract base class defines interface without full implementation
  • Polymorphism: Different account types implement withdraw differently
  • Information Hiding: Internal implementation details are hidden from users

OOP Quality Metrics and Design Principles

Evaluating software quality through object-oriented metrics

Essential OOP Metrics

These metrics help assess the quality and maintainability of object-oriented programs:

Cohesion

Definition: How closely related the responsibilities of a class are

High Cohesion: Class has a single, well-defined purpose

Low Cohesion: Class handles multiple unrelated tasks

Coupling

Definition: Degree of interdependence between classes

Low Coupling: Classes are independent and interact through interfaces

High Coupling: Changes in one class affect many others

Encapsulation

Definition: Bundling data with methods that operate on that data

Good Encapsulation: Private attributes with public accessors

Poor Encapsulation: Public attributes allowing direct modification

Coupling Accounts depend only on abstract base class Low - Interface-based interaction

Complete Banking System Implementation

from abc import ABC, abstractmethod
from datetime import datetime

class BankOperation(ABC):
    """Interface for all bank operations."""
    
    @abstractmethod
    def execute(self):
        pass
    
    @abstractmethod
    def get_description(self):
        pass

class DepositOperation(BankOperation):
    def __init__(self, account, amount):
        self.account = account
        self.amount = amount
    
    def execute(self):
        return self.account.deposit(self.amount)
    
    def get_description(self):
        return f"Deposit £{self.amount}"

class BankingSystem:
    """Main banking system using abstraction."""
    
    def __init__(self):
        self._accounts = {}
        self._operations = []
    
    def add_account(self, account):
        self._accounts[account._account_number] = account
    
    def process_operation(self, operation):
        result = operation.execute()
        if result:
            self._operations.append({
                'operation': operation.get_description(),
                'success': True
            })
        return result

# Demonstration
system = BankingSystem()
savings = SavingsAccount("David", 2000)
system.add_account(savings)

operations = [
    DepositOperation(savings, 500),
    DepositOperation(savings, 1000)
]

print("Processing operations:")
for op in operations:
    success = system.process_operation(op)
    status = "✓" if success else "✗"
    print(f"{status} {op.get_description()}")
